{{- /* Returns the built Scholarship Pages that quote an Episode as a Trigger Claim. Takes a dict: episode the Episode Number sources a list of trigger sources, each a dict: base section path of the pages, e.g. "/scholarship" triggers map of page slug to its Trigger Claim list (#92 shape) Timeline Entries come from timeline-entries-for-episode.html. Draft pages are not built without -D, so GetPage skips them. */ -}} {{- $episode := int .episode -}} {{- $pages := slice -}} {{- range .sources -}} {{- $base := .base -}} {{- range $slug, $triggers := .triggers -}} {{- $hits := where $triggers "episode" $episode -}} {{- if $hits -}} {{- with site.GetPage (path.Join $base $slug) -}} {{- $pages = $pages | append . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return sort ($pages | uniq) "Title" -}}
