{{- /* The Timeline minimap (#167 §1–§3): the whole Timeline drawn to time, a fixed rail in the inline-start gutter on wide screens, and on narrow screens a sticky "You are in" bar whose Contents button opens the same minimap as a popover. Era bands are sized by their year span and each placed entry is a tick at its year, so Hugo lays it all out; nothing is measured in the page. Takes the eras from timeline-eras.html. "You are here" is CSS. One hidden list of entry links is the only scrollspy (scroll-target-group); it stays rendered while the popover is closed, so the bar tracks the era with no geometry. The generated :has() rules below key every current-state style on its :target-current link: the caption label, the bar's era, the shaded band, the extended tick and the entry's narrated setting (#170). Only entries are scroll targets, so the highlight never alternates between eras and entries. timeline-minimap.ts mirrors the current tick into aria-current on scrollend. */ -}} {{- $eras := . -}} {{- $placedLabel := (partial "timeline-labels.html").placed -}} {{- $start := float (index $eras 0).from -}} {{- $span := sub (float (index $eras (sub (len $eras) 1)).to) $start -}} {{- /* One flat model: eras with their ids, entries with their era, strip position and caption date. */ -}} {{- $bands := slice -}} {{- $ticks := slice -}} {{- range $eras -}} {{- $eraId := partial "timeline-era-id.html" . -}} {{- $bands = $bands | append (dict "id" $eraId "label" .label "span" (sub .to .from) "empty" (not .items)) -}} {{- $prevYear := "" -}} {{- $col := 0 -}} {{- range .items -}} {{- $p := .page.Params -}} {{- $year := int .year -}} {{- /* Same-year entries step sideways, so each tick stays visible and clickable. */ -}} {{- $col = cond (eq (string $year) $prevYear) (add $col 1) 0 -}} {{- $prevYear = string $year -}} {{- $date := "" -}} {{- if eq .by "tradition" -}} {{- $date = printf "%s (%s)" $p.tradition.display (lower $placedLabel) -}} {{- else -}} {{- $date = $p.critical.display -}} {{- end -}} {{- $pos := div (sub (float $year) $start) $span -}} {{- /* A narrated setting with years: a span at its years and an arc from its middle to the tick. */ -}} {{- $setting := dict -}} {{- with $p.setting -}} {{- if isset . "from" -}} {{- $from := div (sub (float .from) $start) $span -}} {{- $to := div (sub (float (.to | default .from)) $start) $span -}} {{- $mid := div (add $from $to) 2 -}} {{- $setting = dict "from" $from "to" $to "arcFrom" (math.Min $mid $pos) "arcTo" (math.Max $mid $pos) -}} {{- end -}} {{- end -}} {{- $ticks = $ticks | append (dict "setting" $setting "id" .page.File.ContentBaseName "era" $eraId "title" .page.Title "date" $date "placed" (eq .by "tradition") "pos" $pos "col" $col) -}} {{- end -}} {{- end -}}
