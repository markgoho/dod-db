{{- /* The Timeline minimap (#167 §1–§3): the whole Timeline drawn to time, a fixed rail in the inline-start gutter on wide screens, and on narrow screens a sticky "You are in" bar whose Contents button opens the same minimap as a popover. On wide screens a pill fixed over the axis names the current era. Era bands are sized by their year span and each placed entry is a tick at its year, so Hugo lays it all out; nothing is measured in the page. Takes the eras from timeline-eras.html. "You are here" is CSS. One hidden list of entry links is the only scrollspy (scroll-target-group); it stays rendered while the popover is closed, so the bar tracks the era with no geometry. The generated :has() rules below set one flag, --is-current: 1, on whatever its :target-current link makes current: the caption label, the bar's era, the band, the tick, the entry's narrated setting (#170) and the entry on the axis. timeline-minimap.css says what the flag shows. Only entries are scroll targets, so the highlight never alternates between eras and entries. timeline-minimap.ts (loaded from the list layout's footer scripts) mirrors the current tick into aria-current. */ -}} {{- $eras := . -}} {{- $labels := partial "timeline-labels.html" -}} {{- $placedLabel := $labels.placed -}} {{- $start := float (index $eras 0).from -}} {{- $years := sub (float (index $eras (sub (len $eras) 1)).to) $start -}} {{- /* A tick is a dot at its year; --tick-pitch (dot plus gap) is one dot's place: 0.04 of the shortest strip (15px of about 380px, in a 360×640 popover) and less on the rail (18px of 787px at 1728×1000; measured), so 0.04 here. Ticks closer than that may overlap, as Creation (4004 BCE) and Adam (4000 BCE) did, so each takes the lowest slot (--tick-slot) none of them has. Slots are per column: a tick placed by a tradition's date (the dashed dot, and the diamond of a moved Tradition Date, #214) sits in its own thin column beside the column of scholarly dots, so the two kinds never compete for a slot. A row fans out from its column's centre: slot 0 on the centre line, then a pitch to one side, a pitch to the other, two pitches to the first side, and so on (--tick-lane: 0, 1, -1, 2, -2…). Each tick also carries its column neighbours' years (--tick-prev, --tick-next; -1 and 2 when there is none), so the CSS can pull a row back to its centre line where the dots already clear each other vertically. Each tick carries its era's edges too, so the CSS keeps its dot inside its band: a dot at an era's edge, like Creation's and Adam's at 4000 BCE, is nudged in off the line (#175), never more than its own size. */ -}} {{- $pitchShare := 0.04 -}} {{- /* One flat model: eras with their ids, entries with their era, strip position, column, row slot and caption date. */ -}} {{- $bands := slice -}} {{- $ticks := slice -}} {{- range $eras -}} {{- $eraId := partial "timeline-era-id.html" . -}} {{- $eraStart := div (sub (float .from) $start) $years -}} {{- $eraEnd := div (sub (float .to) $start) $years -}} {{- $bands = $bands | append (dict "id" $eraId "label" .label "start" $eraStart "end" $eraEnd "empty" (not .items)) -}} {{- range .items -}} {{- $p := .page.Params -}} {{- $date := "" -}} {{- $isMoved := eq .by "moved" -}} {{- if eq .by "tradition" -}} {{- $date = printf "%s (%s)" $p.tradition.display (lower $placedLabel) -}} {{- else if $isMoved -}} {{- $date = printf "%s (%s)" $p.tradition.display $labels.movedShort -}} {{- else -}} {{- $date = $p.critical.display -}} {{- end -}} {{- $column := cond (or $isMoved (eq .by "tradition")) "tradition" "critical" -}} {{- $at := div (sub (float .year) $start) $years -}} {{- $taken := slice -}} {{- range $ticks -}} {{- if and (eq .column $column) (lt (math.Abs (sub $at .at)) $pitchShare) -}} {{- $taken = $taken | append .slot -}} {{- end -}} {{- end -}} {{- $slot := 0 -}} {{- range seq (len $taken) -}} {{- if in $taken $slot -}}{{- $slot = add $slot 1 -}}{{- end -}} {{- end -}} {{- /* Fan out: 0, 1, -1, 2, -2… */ -}} {{- $lane := cond (modBool $slot 2) (div $slot -2) (div (add $slot 1) 2) -}} {{- /* A narrated setting with years: a span at its years and an arc from its middle to the tick's dot. */ -}} {{- $setting := dict -}} {{- with and (not $isMoved) $p.setting -}} {{- if isset . "from" -}} {{- $settingStart := div (sub (float .from) $start) $years -}} {{- $settingEnd := div (sub (float (.to | default .from)) $start) $years -}} {{- $mid := div (add $settingStart $settingEnd) 2 -}} {{- $setting = dict "start" $settingStart "end" $settingEnd "mid" $mid -}} {{- end -}} {{- end -}} {{- $ticks = $ticks | append (dict "setting" $setting "id" (cond $isMoved (printf "%s-tradition" .page.File.ContentBaseName) .page.File.ContentBaseName) "moved" $isMoved "era" $eraId "eraStart" $eraStart "eraEnd" $eraEnd "title" .page.Title "date" $date "placed" (eq .by "tradition") "column" $column "at" $at "slot" $slot "lane" $lane) -}} {{- end -}} {{- end -}} {{- /* Each tick's nearest neighbours in its column, above and below (a tick at the same year is both). */ -}} {{- $withNeighbours := slice -}} {{- range $i, $t := $ticks -}} {{- $prev := -1.0 -}} {{- $next := 2.0 -}} {{- range $j, $o := $ticks -}} {{- if and (ne $i $j) (eq $o.column $t.column) -}} {{- if le $o.at $t.at -}} {{- $prev = math.Max $prev $o.at -}} {{- end -}} {{- if ge $o.at $t.at -}} {{- $next = math.Min $next $o.at -}} {{- end -}} {{- end -}} {{- end -}} {{- $withNeighbours = $withNeighbours | append (merge $t (dict "prev" $prev "next" $next)) -}} {{- end -}} {{- $ticks = $withNeighbours -}} {{- /* The lens's hover targets (timeline-minimap.css): this many still slices down the strip. The lens moves the year at its focus a little (towards the middle), so each slice focuses the year the lens puts at the slice's own middle, found by bisection: the bulge is centred under the pointer. The gain and width are timeline-minimap.css's --lens-max-gain and --lens-width (src/hugo/timeline-lens.ts; its test checks all three agree). */ -}} {{- $lensSlices := 64 -}} {{- $lensGain := 2.5 -}} {{- $lensWidth := 0.06 -}} {{- $sliceFoci := slice -}} {{- range $i := seq $lensSlices -}} {{- $target := div (sub (float $i) 0.5) $lensSlices -}} {{- $lo := 0.0 -}} {{- $hi := 1.0 -}} {{- range seq 30 -}} {{- $f := div (add $lo $hi) 2 -}} {{- $aw := mul $lensGain $lensWidth -}} {{- $n := add 1 (mul $aw (add (div (sub 1 $f) (add $lensWidth (sub 1 $f))) (div $f (add $lensWidth $f)))) -}} {{- $y := div (add $f (div (mul $aw $f) (add $lensWidth $f))) $n -}} {{- if lt $y $target -}} {{- $lo = $f -}} {{- else -}} {{- $hi = $f -}} {{- end -}} {{- end -}} {{- $sliceFoci = $sliceFoci | append (div (add $lo $hi) 2) -}} {{- end -}}
