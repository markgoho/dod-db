{{- /* The id of one Timeline era, shared by the axis and the minimap's era links (#167 §1). Takes an era from timeline-eras.html; returns its period's id ("era-bronze-age", #256) or, for a span with no period, its start as a year: "era-5000-bce", "era-1-ce". */ -}} {{- $id := "" -}} {{- with .period -}} {{- $id = printf "era-%s" . -}} {{- else -}} {{- $from := int .from -}} {{- if lt $from 0 -}} {{- $id = printf "era-%d-bce" (sub 0 $from) -}} {{- else -}} {{- $id = printf "era-%d-ce" (cond (eq $from 0) 1 $from) -}} {{- end -}} {{- end -}} {{- return $id -}}
