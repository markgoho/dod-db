{{- /* Where each published Timeline Entry sits (#163 §1). Takes the entry pages; returns dict: byId entry id → entry page placed axis entries in date order, each dict {key, page, year, by}: year the placement year (Critical Date sort, or Tradition Date) by "critical", or "tradition" for an undated entry placed by its Tradition Date in the tradition lane only unplaced undated entries with no Tradition Date, by title ordered every entry page: placed, then unplaced The only source of order: the page body and its JSON-LD both use it. */ -}} {{- $entries := . -}} {{- $byId := dict -}} {{- range $entries -}} {{- $byId = merge $byId (dict .File.ContentBaseName .) -}} {{- end -}} {{- /* An anchored entry ("exilic") takes its years from its anchor's sourced date; placement never invents a year. Sort by placement year, then events first, then title. */ -}} {{- $placed := slice -}} {{- $unplaced := slice -}} {{- range $entry := $entries -}} {{- $c := .Params.critical -}} {{- $t := .Params.tradition -}} {{- $place := dict -}} {{- if $c -}} {{- $from := $c.from -}} {{- with $c.anchor -}} {{- $a := index $byId . -}} {{- if not $a -}} {{- errorf "Timeline: %s anchors to %q, which is not a published entry" $entry.File.Path . -}} {{- end -}} {{- $from = $a.Params.critical.from -}} {{- end -}} {{- $place = dict "year" ($c.sort | default $from) "by" "critical" -}} {{- else if $t -}} {{- $place = dict "year" $t.from "by" "tradition" -}} {{- end -}} {{- with $place -}} {{- $key := printf "%07d|%d|%s" (add (int .year) 1000000) (cond (eq $entry.Params.entryKind "event") 0 1) (lower $entry.Title) -}} {{- $placed = $placed | append (merge . (dict "key" $key "page" $entry)) -}} {{- else -}} {{- $unplaced = $unplaced | append $entry -}} {{- end -}} {{- end -}} {{- $placed = sort $placed "key" -}} {{- $unplaced = sort $unplaced "Title" -}} {{- $ordered := slice -}} {{- range $placed -}} {{- $ordered = $ordered | append .page -}} {{- end -}} {{- range $unplaced -}} {{- $ordered = $ordered | append . -}} {{- end -}} {{- return dict "byId" $byId "placed" $placed "unplaced" $unplaced "ordered" $ordered -}}
