{{- /* Returns the built Scholarship Pages whose `related` names a term. Takes a dict: page the Book, Tag or Topic term page taxonomy "books", "tags" or "topics" (the `related` key) A term matches by scholarship-related-term.html. Draft pages are not built without -D. */ -}} {{- $term := .page -}} {{- $taxonomy := .taxonomy -}} {{- $pages := slice -}} {{- with site.GetPage "/scholarship" -}} {{- range .RegularPages.ByTitle -}} {{- $page := . -}} {{- range index (.Params.related | default dict) $taxonomy -}} {{- with partial "scholarship-related-term.html" (dict "taxonomy" $taxonomy "term" .) -}} {{- if eq .RelPermalink $term.RelPermalink -}} {{- $pages = $pages | append $page -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $pages -}}
