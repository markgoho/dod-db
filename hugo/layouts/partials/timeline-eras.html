{{- /* The Timeline axis's eras: the age track of hugo/data/timeline/periods.yaml (#255), so the axis is divided by cited historical periods, not round years (#256). Takes the placed entries from timeline-placement.html (each with a placement year). The axis starts at the earliest placement year rounded down to $wide and ends after the latest, rounded up to $narrow. Each age period is an era, clipped to the axis; the span before the first one (and after the last, if it ends) is an era named only by its years. The age periods must meet end to start, oldest first: a gap or overlap fails the build. Returns a slice of dict {label, dates, period, from, to, compressed, items, opens}, oldest first: label is the period's name or, for a span with no period, its years ("5000–3000 BCE"); dates is the period's display dates ("" for a span); period its id. compressed marks the span before the first period, which the minimap may draw short. items keep placement order. Empty when nothing is placed. opens lists the temple-track periods that start where a period era starts ("Exile" with Babylonian rule), so the axis names them (#276). */ -}} {{- $placed := . -}} {{- $wide := 1000 -}}{{- /* the axis starts on a multiple of this */ -}} {{- $narrow := 500 -}}{{- /* the axis ends on a multiple of this */ -}} {{- $eras := slice -}} {{- with $placed -}} {{- $years := slice -}} {{- range . -}} {{- $years = $years | append (int .year) -}} {{- end -}} {{- $start := int (mul (math.Floor (div (float (math.Min $years)) $wide)) $wide) -}} {{- $end := int (mul (add (math.Floor (div (float (math.Max $years)) $narrow)) 1) $narrow) -}} {{- $ages := sort (where (hugo.Data.timeline.periods | default slice) "track" "age") "from" -}} {{- $temples := where (hugo.Data.timeline.periods | default slice) "track" "temple" -}} {{- $bounds := slice -}} {{- $at := $start -}} {{- range $i, $p := $ages -}} {{- if and (gt $i 0) (ne (int (index $ages (sub $i 1)).to) (int $p.from)) -}} {{- errorf "timeline-eras: age period %q does not start where %q ends" $p.id (index $ages (sub $i 1)).id -}} {{- end -}} {{- $from := math.Max (int $p.from) $start | int -}} {{- $to := math.Min (int ($p.to | default $end)) $end | int -}} {{- if lt $from $to -}} {{- if lt $at $from -}} {{- $bounds = $bounds | append (dict "from" $at "to" $from "compressed" (eq (len $bounds) 0)) -}} {{- end -}} {{- $bounds = $bounds | append (dict "from" $from "to" $to "period" $p.id "label" $p.label "dates" $p.display) -}} {{- $at = $to -}} {{- end -}} {{- end -}} {{- if lt $at $end -}} {{- $bounds = $bounds | append (dict "from" $at "to" $end "compressed" (eq (len $bounds) 0)) -}} {{- end -}} {{- range $bounds -}} {{- $from := .from -}} {{- $to := .to -}} {{- /* A span with no period is named by its years, [from, to): -5000..-3000 is "5000–3000 BCE"; -500..0 is "500–1 BCE". */ -}} {{- $label := .label | default "" -}} {{- if not $label -}} {{- if lt $from 0 -}} {{- if gt $to 0 -}} {{- $label = printf "%d BCE–%d CE" (sub 0 $from) $to -}} {{- else -}} {{- $label = printf "%d–%d BCE" (sub 0 $from) (cond (eq $to 0) 1 (sub 0 $to)) -}} {{- end -}} {{- else -}} {{- $label = printf "%d–%d CE" (cond (eq $from 0) 1 $from) $to -}} {{- end -}} {{- end -}} {{- $items := slice -}} {{- range $placed -}} {{- if and (ge (int .year) $from) (lt (int .year) $to) -}} {{- $items = $items | append . -}} {{- end -}} {{- end -}} {{- $opens := slice -}} {{- if .period -}} {{- $opens = where $temples "from" $from -}} {{- end -}} {{- $eras = $eras | append (dict "label" $label "dates" (.dates | default "") "period" (.period | default "") "from" $from "to" $to "compressed" (.compressed | default false) "items" $items "opens" $opens) -}} {{- end -}} {{- end -}} {{- return $eras -}}
