{{- /* The Timeline axis's era bands (#163 §2). Takes the placed entries from timeline-placement.html (each with a placement year). The axis starts at the earliest placement year rounded down to $wide and ends after the latest, rounded up to $narrow. Bands before $split (2000 BCE) are $wide (1000) years, after it $narrow (500). Returns a slice of dict {label, from, to, items}, oldest first; items keep placement order. Empty when nothing is placed. */ -}} {{- $placed := . -}} {{- $wide := 1000 -}}{{- /* band width before $split */ -}} {{- $narrow := 500 -}}{{- /* band width from $split on */ -}} {{- $split := -2000 -}}{{- /* 2000 BCE: where bands narrow */ -}} {{- $eras := slice -}} {{- with $placed -}} {{- $years := slice -}} {{- range . -}} {{- $years = $years | append (int .year) -}} {{- end -}} {{- $start := int (mul (math.Floor (div (float (math.Min $years)) $wide)) $wide) -}} {{- $end := int (mul (add (math.Floor (div (float (math.Max $years)) $narrow)) 1) $narrow) -}} {{- $bounds := slice -}} {{- if lt $start $split -}} {{- /* The last wide band starts one band before $split; none starts at or after $end. */ -}} {{- range seq $start $wide (math.Min (sub $split $wide) (sub $end 1) | int) -}} {{- $bounds = $bounds | append (dict "from" . "to" (add . $wide)) -}} {{- end -}} {{- end -}} {{- $late := math.Max $start $split | int -}} {{- if gt $end $late -}} {{- range seq $late $narrow (sub $end $narrow) -}} {{- $bounds = $bounds | append (dict "from" . "to" (add . $narrow)) -}} {{- end -}} {{- end -}} {{- range $bounds -}} {{- $from := .from -}} {{- $to := .to -}} {{- /* [from, to): -1500..-1000 is "1500–1000 BCE"; -500..0 is "500–1 BCE". */ -}} {{- $label := "" -}} {{- if lt $from 0 -}} {{- $label = printf "%d–%d BCE" (sub 0 $from) (cond (eq $to 0) 1 (sub 0 $to)) -}} {{- else -}} {{- $label = printf "%d–%d CE" (cond (eq $from 0) 1 $from) $to -}} {{- end -}} {{- $items := slice -}} {{- range $placed -}} {{- if and (ge (int .year) $from) (lt (int .year) $to) -}} {{- $items = $items | append . -}} {{- end -}} {{- end -}} {{- $eras = $eras | append (dict "label" $label "from" $from "to" $to "items" $items) -}} {{- end -}} {{- end -}} {{- return $eras -}}
