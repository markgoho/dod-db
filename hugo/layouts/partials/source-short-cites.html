{{- /* Returns short cites for a list of External Source ids, in first-cited order: a map of id to its author-date key, e.g. "Meyers 2005", "Coogan and Chapman 2019" or "Di Lazzaro et al. 2020". Authors come from the registry's author field, a list of names joined by ", " and " and ": "(ed.)"/"(eds.)" is dropped; one name gives its surname, two give "A and B", and three or more give the first surname and "et al.", as does a line ending "et al." or "and others". A surname is a name's last word after any "Jr."/"Sr." suffix, with any particles before it ("Di Lazzaro", "van Deventer"). A source without an author falls back to its title up to the first colon. A registry shortAuthor replaces the whole derived name, for a line the rule can't read (#225): a compound surname without a particle ("Davis Bledsoe") or a regnal name, which has no surname ("John Paul II"). The partial doesn't guess regnal numerals: "II" can't be told from a generational "III". Two sources sharing a key get letter suffixes ("Hayes 2006a", "Hayes 2006b") in the order given, scoped to the page, as in an author-date reference list. Used by Scholarship Pages and the Timeline. */ -}} {{- $registry := index hugo.Data "external-sources" -}} {{- $ids := . -}} {{- $particles := slice "da" "de" "del" "della" "der" "di" "du" "la" "le" "van" "von" "Da" "De" "Del" "Della" "Der" "Di" "Du" "La" "Le" "Van" "Von" -}} {{- $base := dict -}} {{- range $ids -}} {{- $s := index $registry . -}} {{- $key := "" -}} {{- if $s.shortAuthor -}} {{- $key = $s.shortAuthor -}} {{- else if $s.author -}} {{- $line := replaceRE `\s*\(eds?\.\)` "" $s.author -}} {{- $etAl := findRE `,?\s+(et al\.?|and others)$` $line 1 -}} {{- $line = replaceRE `,?\s+(et al\.?|and others)$` "" $line -}} {{- $line = replaceRE `,?\s+(Jr|Sr)\.?(,|\s+and\s|$)` "$2" $line -}} {{- $surnames := slice -}} {{- range split (replaceRE `,\s+and\s+|,\s+|\s+and\s+` "|" $line) "|" -}} {{- $words := split (trim . " ") " " -}} {{- $i := sub (len $words) 1 -}} {{- range $n := seq (len $words) -}} {{- if and (eq $i (sub (len $words) $n)) (gt $i 0) -}} {{- if in $particles (index $words (sub $i 1)) -}} {{- $i = sub $i 1 -}} {{- end -}} {{- end -}} {{- end -}} {{- $surnames = $surnames | append (delimit (after $i $words) " ") -}} {{- end -}} {{- if or $etAl (gt (len $surnames) 2) -}} {{- $key = printf "%s et al." (index $surnames 0) -}} {{- else -}} {{- $key = delimit $surnames " and " -}} {{- end -}} {{- else -}} {{- $key = trim (index (split $s.title ":") 0) " " -}} {{- end -}} {{- with $s.year -}} {{- $key = printf "%s %v" $key . -}} {{- end -}} {{- $base = merge $base (dict . $key) -}} {{- end -}} {{- $letters := split "abcdefghijklmnopqrstuvwxyz" "" -}} {{- $cites := dict -}} {{- range $id := $ids -}} {{- $key := index $base $id -}} {{- $peers := slice -}} {{- range $ids -}} {{- if eq (index $base .) $key -}} {{- $peers = $peers | append . -}} {{- end -}} {{- end -}} {{- if gt (len $peers) 1 -}} {{- range $i, $peer := $peers -}} {{- if eq $peer $id -}} {{- $key = printf "%s%s" $key (index $letters $i) -}} {{- end -}} {{- end -}} {{- end -}} {{- $cites = merge $cites (dict $id $key) -}} {{- end -}} {{- return $cites -}}
