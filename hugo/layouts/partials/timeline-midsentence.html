The stored display, and a card that shows it on its own line, keep the capital. Takes the string. */ -}} {{- $s := . -}} {{- range slice "About" "After" "Before" "Late" "Presumably" "Probably" "Toward" -}} {{- if strings.HasPrefix $s (printf "%s " .) -}} {{- $s = printf "%s%s" (lower .)
